# ENV settings for coin-dashboard-bot
# New / updated settings for TZ 2026-09-15

# ============================================================
# MAGNET / VOLUME PROFILE / VPOC
# ============================================================

# Exponential decay half-life for volume-profile candle weight.
# 7 = after 7 days, the weight of a candle is approximately 50%.
MAGNET_VP_DECAY_HALF_LIFE_DAYS=7

# Short VPOC calculation window.
# Short VPOC uses the most recent N days and is reported as vpoc_short.
MAGNET_VP_SHORT_WINDOW_DAYS=7


# ============================================================
# ORDER BOOK SAMPLING
# ============================================================

# Number of order-book snapshots during one /coin request.
# 3 means: snapshot -> wait -> snapshot -> wait -> snapshot.
BOOK_SAMPLE_COUNT=3

# Delay between order-book snapshots, in seconds.
# 4 means approximately 8 seconds total between the first and last sample.
BOOK_SAMPLE_INTERVAL_SECONDS=4

# Price tolerance for considering walls to be in the same price bucket.
# 0.10 = 0.10%.
BOOK_WALL_TOLERANCE_PCT=0.10

# Allowed change in wall notional for a wall to remain "stable".
# 30 = +/-30% from the reference wall size.
BOOK_WALL_SIZE_TOLERANCE_PCT=30


# ============================================================
# RELAXDEV STORAGE BACKUP
# ============================================================

# Minimum interval between automatic SQLite backups.
# 3600 seconds = 1 hour.
STORAGE_BACKUP_INTERVAL_SECONDS=3600

# Number of recent backup files to retain.
STORAGE_BACKUP_KEEP=24

# Folder/path in RelaxDev Storage where backups are stored.
STORAGE_BACKUP_PATH=magnet-backups


# ============================================================
# STORAGE API KEY
# ============================================================

# Do NOT normally add STORAGE_API_KEY to this file.
# RelaxDev injects the Storage API key when Storage is configured.
